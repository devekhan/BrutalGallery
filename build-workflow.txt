name: Build Debug APK

on:
  push:
    branches: [ main, master ]
  workflow_dispatch:

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - name: Get the code
        uses: actions/checkout@v4

      - name: Install Java 17
        uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: '17'

      - name: Download the bundled fonts (Archivo Black + Space Mono)
        run: |
          mkdir -p app/src/main/res/font
          BASE=https://raw.githubusercontent.com/google/fonts/main/ofl
          curl -fL -o app/src/main/res/font/archivo_black.ttf       $BASE/archivoblack/ArchivoBlack-Regular.ttf
          curl -fL -o app/src/main/res/font/space_mono_regular.ttf  $BASE/spacemono/SpaceMono-Regular.ttf
          curl -fL -o app/src/main/res/font/space_mono_bold.ttf     $BASE/spacemono/SpaceMono-Bold.ttf

      - name: Install Gradle
        uses: gradle/actions/setup-gradle@v4
        with:
          gradle-version: 8.10.2

      - name: Build debug APK
        run: gradle assembleDebug --console=plain > build.log 2>&1

      - name: Show the errors (only runs if the build failed)
        if: failure()
        run: |
          echo "=============== KOTLIN ERRORS ==============="
          grep -E "^e: " build.log | head -40 || true
          echo "=============== WHAT WENT WRONG ==============="
          grep -A 12 "What went wrong" build.log | head -40 || true

      - name: Save the full log (only if the build failed)
        if: failure()
        uses: actions/upload-artifact@v4
        with:
          name: build-log
          path: build.log

      - name: Upload APK
        uses: actions/upload-artifact@v4
        with:
          name: BrutalGallery-debug-apk
          path: app/build/outputs/apk/debug/app-debug.apk
          if-no-files-found: error
